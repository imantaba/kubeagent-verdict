func pullFinding(ns, pod, issue, image string) diagnose.Finding {
	return diagnose.Finding{
		Pod:      ns + "/" + pod,

		Evidence: `Failed to pull image "` + image + `"`,

	}
}

		Namespace: ns, Name: name, Kind: "Deployment",

		Findings: []diagnose.Finding{pullFinding(ns, pod, issue, image)},
		Pods:     []inventory.PodRow{{Name: pod, Node: node}},
	}
}

func nodeWith(name string, unschedulable bool, conds []corev1.NodeCondition, taints []corev1.Taint) *corev1.Node {
	n := &corev1.Node{ObjectMeta: metav1.ObjectMeta{Name: name}}
	n.Spec.Unschedulable = unschedulable
	n.Spec.Taints = taints
	n.Status.Conditions = conds
	return n
}

func ready(status corev1.ConditionStatus, reason, msg string) corev1.NodeCondition {
	return corev1.NodeCondition{Type: corev1.NodeReady, Status: status, Reason: reason, Message: msg}
}

func pvcWith(name string, phase corev1.PersistentVolumeClaimPhase, class, volume string) *corev1.PersistentVolumeClaim {
	p := &corev1.PersistentVolumeClaim{ObjectMeta: metav1.ObjectMeta{Namespace: "db", Name: name}}
	p.Spec.VolumeName = volume
	if class != "" {
		c := class
		p.Spec.StorageClassName = &c
	}
	p.Status.Phase = phase
	return p
}

type readJSON struct {
	Label   string `json:"label"`
	Content string `json:"content"`
}

// captureBundle is the eight reads the prompt shows, in gather order:
// nodes, then PVCs, then events. describeNode, describePVC and
// formatEvents live in reader.go (same package; reader.go:238, :250, :312
// at v1.24.0); open it for their exact signatures and call them with the
// objects captureReads holds so the content bytes are the gather's own.
func captureBundle(reads hypothesis.Reads) []readJSON {
	return []readJSON{
		{"describe node /worker-1", describeNode(reads.Nodes["worker-1"])},
		{"describe node /worker-2", describeNode(reads.Nodes["worker-2"])},
		{"describe pvc db/aux-0", describePVC(reads.PVCs["db/aux-0"])},
		{"describe pvc db/data-0", describePVC(reads.PVCs["db/data-0"])},
		{"describe pvc db/cache-0", describePVC(reads.PVCs["db/cache-0"])},
		{"events img/" + pods["img/one"], formatEvents("img", pods["img/one"], reads.Events["img/"+pods["img/one"]])},
		{"events img/" + pods["img/five"], formatEvents("img", pods["img/five"], nil)},
		{"describe node /worker-7", "read failed: " + reads.Failed["node/worker-7"]},
	}
}

func candidatesJSON(w inventory.Workload, r hypothesis.Result) []candidateJSON {
	out := make([]candidateJSON, 0, len(w.RootCauseTrace))
	for _, h := range w.RootCauseTrace {
		c := candidateJSON{Cause: h.Cause, Verdict: string(h.Verdict), Reason: h.Reason}
		for _, d := range r.Decisions {
			if d.Candidate.Cause == h.Cause {
				c.FreshReadOutcome, c.FreshReadEvidence = string(d.Outcome), d.Evidence
			}
		}
		out = append(out, c)
	}
	return out
}

func resultToJSON(r hypothesis.Result) resultJSON {
	out := resultJSON{Decided: r.Decided, Cause: r.Cause, Outcome: string(r.Outcome),
		Evidence: r.Evidence, GroupKey: r.GroupKey, GroupText: r.GroupText,
		Decisions: []decisionJSON{}}
	for _, d := range r.Decisions {
		out.Decisions = append(out.Decisions, decisionJSON{d.Candidate.Cause, string(d.Outcome), d.Evidence})
	}
	return out
}

// clusterJSON, lineJSON, summaryJSON, serviceIssuesJSON and findingsJSON
// mirror kubeagent's own field shapes into the same JSON key names
// contract/golden/input.json already uses (snake_case, "type" for a
// service issue's kind). Go's json encoder sorts a map's keys, so the
// insertion order here does not matter.
func clusterJSON(c clusterhealth.ClusterHealth) map[string]any {
	nodeIssues := c.NodeIssues
	if nodeIssues == nil {
		nodeIssues = []string{}
	}
	systemIssues := c.SystemIssues
	if systemIssues == nil {
		systemIssues = []string{}
	}
	return map[string]any{
		"degraded":      c.Verdict == "Degraded",
		"nodes_ready":   c.NodesReady,
		"nodes_total":   c.NodesTotal,
		"node_issues":   nodeIssues,
		"system_issues": systemIssues,
	}
}

func lineJSON(l resources.Line) map[string]any {
	return map[string]any{
		"allocatable":  l.Allocatable,
		"requests":     l.Requests,
		"requests_pct": l.RequestsPct,
		"limits":       l.Limits,
		"limits_pct":   l.LimitsPct,
	}
}

func summaryJSON(s *resources.Summary) map[string]any {
	if s == nil {
		return map[string]any{}
	}
	return map[string]any{
		"cpu":               lineJSON(s.CPU),
		"memory":            lineJSON(s.Memory),
		"metrics_available": s.MetricsAvailable,
	}
}

func serviceIssuesJSON(issues []svchealth.Issue) []map[string]any {
	out := make([]map[string]any, 0, len(issues))
	for _, iss := range issues {
		out = append(out, map[string]any{
			"namespace": iss.Namespace,
			"name":      iss.Name,
			"type":      iss.Type,
			"detail":    iss.Detail,
		})
	}
	return out
}

// findingsJSON converts a workload's findings into the JSON shape today's
// golden uses. next_step/command are remediation's own deterministic
// suggestion for the finding's Issue, exactly as the prompt's "suggested
// fix" line reports it: explain.go's suggestionFor masks f.Pod to <pod>
// before calling remediation.For whenever the finding's pod differs from
// the owning workload's own namespace/name, and this mirrors that rule so
// input.json's command values match the prompt's masked run: lines byte
// for byte.
func findingsJSON(w inventory.Workload, findings []diagnose.Finding) []map[string]any {
	out := make([]map[string]any, 0, len(findings))
	for _, f := range findings {
		if f.Pod != "" && f.Pod != w.Namespace+"/"+w.Name {
			if i := strings.IndexByte(f.Pod, '/'); i >= 0 {
				f.Pod = f.Pod[:i+1] + "<pod>"
			} else {
				f.Pod = "<pod>"
			}
		}
		s := remediation.For(f)
		out = append(out, map[string]any{
			"issue":     f.Issue,
			"reason":    f.Reason,
			"evidence":  f.Evidence,
			"log_cause": f.LogCause,
			"next_step": s.NextStep,
			"command":   s.Command,
		})
	}
	return out
}

func writeFile(t *testing.T, dir, name string, data []byte) {
	t.Helper()
	if err := os.WriteFile(filepath.Join(dir, name), data, 0o644); err != nil {
		t.Fatal(err)
	}
}

func TestCaptureVerdictGolden(t *testing.T) {
	dir := os.Getenv("KV_GOLDEN_DIR")
	if dir == "" {
		t.Skip("KV_GOLDEN_DIR not set")
	}
	if err := os.MkdirAll(dir, 0o755); err != nil {
		t.Fatal(err)
	}

	wl := captureWorkloads()
	down := []clusterhealth.DownNode{
		{Name: "worker-1", Reason: "NotReady"},
		{Name: "worker-2", Reason: "no kubelet lease"},
		{Name: "worker-3", Reason: "kubelet not heartbeating"},
		{Name: "worker-5", Reason: "NotReady"},
		{Name: "worker-6", Reason: "NotReady"},
		{Name: "worker-7", Reason: "NotReady"},
	}
	rootcause.Annotate(wl, down)
	rootcause.AnnotatePVC(wl, capturePodPVCs(), capturePVCIssues())
	rootcause.AnnotateRegistry(wl)

	reads := captureReads()
	results := make([]hypothesis.Result, len(wl))
	for i := range wl {
		results[i] = hypothesis.Decide(wl[i], reads)
	}
	shared := hypothesis.Shared(results)

	// The prompt shows the first ten workloads: the gather's cap.
	scoped, scopedResults := wl[:10], results[:10]
	bundle := captureBundle(reads)
	var b strings.Builder
	var trail []string
	spent := 0
	for _, r := range bundle {
		appendRead(&b, &trail, &spent, r.Label, r.Content)
	}
	cluster := goldenCluster()
	summary := goldenSummary()
	issues := goldenServiceIssues()
	prompt := buildVerdictPrompt(cluster, summary, nil, issues, scoped, scopedResults, b.String())
	writeFile(t, dir, "user_message.txt", []byte(prompt))
	writeFile(t, dir, "system_prompt.txt", []byte(verdictSystemPrompt))

	// input.json: today's keys, plus the five new ones. The cluster,
	// summary, platform_line, service_issues and findings blocks use the
	// same conversion the old template used (see the note above the code).
	type workloadJSON struct {
		Namespace      string           `json:"namespace"`
		Name           string           `json:"name"`
		Kind           string           `json:"kind"`
		Ready          int              `json:"ready"`
		Desired        int              `json:"desired"`
		Status         string           `json:"status"`
		Restarts       int              `json:"restarts"`
		Confidence     string           `json:"confidence"`
		Findings       []map[string]any `json:"findings"`
		Candidates     []candidateJSON  `json:"candidates"`
		Decided        bool             `json:"decided"`
		DecidedCause   string           `json:"decided_cause"`
		DecidedOutcome string           `json:"decided_outcome"`
	}
	input := map[string]any{
		"cluster":        clusterJSON(cluster),
		"summary":        summaryJSON(summary),
		"platform_line":  goldenPlatformLine(),
		"service_issues": serviceIssuesJSON(issues),
		"reads":          bundle,
	}
	var wj []workloadJSON
	for i, w := range scoped {
		r := scopedResults[i]
		wj = append(wj, workloadJSON{
			Namespace: w.Namespace, Name: w.Name, Kind: w.Kind,
			Ready: int(w.Ready), Desired: int(w.Desired), Status: w.Status,
			Restarts: int(w.Restarts), Confidence: w.RootCauseConfidence,
			Findings: findingsJSON(w, w.Findings), Candidates: candidatesJSON(w, r),
			Decided: r.Decided, DecidedCause: r.Cause, DecidedOutcome: string(r.Outcome),
		})
	}
	input["workloads"] = wj
	data, err := json.MarshalIndent(input, "", "  ")
	if err != nil {
		t.Fatal(err)
	}
	writeFile(t, dir, "input.json", append(data, '\n'))

	fx := fixtureJSON{
		Kubeagent: map[string]string{"tag": kvTag, "commit": kvCommit},
		Shared:    shared, Reads: bundle,
	}
	if fx.Shared == nil {
		fx.Shared = []string{}
	}
	for i, w := range wl {
		pod, issue, image := firstPod(w)
		fx.Workloads = append(fx.Workloads, fixtureWorkloadJSON{
			Namespace: w.Namespace, Name: w.Name, Pod: pod, Issue: issue, Image: image,
			Candidates: candidatesJSON(w, results[i]), Result: resultToJSON(results[i]),
		})
	}
	data, err = json.MarshalIndent(fx, "", "  ")
	if err != nil {
		t.Fatal(err)
	}
	writeFile(t, dir, "rules_golden.json", append(data, '\n'))
	t.Logf("wrote 4 files under %s; %d workloads in the fixture, %d in the prompt, %d shared lines",
		dir, len(wl), len(scoped), len(shared))
}
