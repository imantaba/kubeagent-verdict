//go:build goldencapture

// Golden capture for kubeagent-verdict. This file lives in
// kubeagent-verdict as contract/capture/kv_capture_test.go.txt and is
// copied into an unpacked archive of kubeagent to run. It never lands
// in kubeagent.
//
// It loads the shared fixture tests/fixtures/gather_fixture.yaml, builds
// the scan-time workloads and a fake clientset from it, and runs
// kubeagent's own pipeline over them, in scan.go's order:
// clusterhealth.Assess, inventory.Prioritize, the three rootcause
// annotators, confidence.Annotate; then local verdict mode's flaggedScope,
// gatherEvidence, hypothesis.Decide and Shared, and buildVerdictPrompt.
// One run writes the golden files and the ten stage dumps
// (tests/fixtures/gather_go/README.md gives their format).
//
// The same run then does it all again over three more fixtures, one per
// mode (kvMode). Each pass writes its own dumps and no golden file:
//   - logs: tests/fixtures/gather_fixture_logs.yaml, whose reads are mostly
//     log reads (the main fixture spends its 8 reads before the gather
//     reaches a crashed container);
//   - cluster: tests/fixtures/gather_fixture_cluster.yaml, a 16-node cluster
//     with every node-health line, many service-issue wordings and a
//     network-policy hint (the prompt's cluster block);
//   - registry: tests/fixtures/gather_fixture_registry.yaml, one workload per
//     registry outcome.
//
// The seven steps. KV is the kubeagent-verdict checkout, SCRATCH a new
// folder outside both repos; run the git commands in the kubeagent repo.
//   1. git status --short    # note what it prints
//      git archive v1.24.0 | gzip > "$SCRATCH/ka.tar.gz"
//      git ls-tree -r v1.24.0 --name-only | wc -l    # the file count
//   2. cp "$KV/tests/fixtures/gather_fixture.yaml" \
//         "$KV/tests/fixtures/gather_fixture_logs.yaml" \
//         "$KV/tests/fixtures/gather_fixture_cluster.yaml" \
//         "$KV/tests/fixtures/gather_fixture_registry.yaml" \
//         "$KV/contract/capture/kv_capture_test.go.txt" "$SCRATCH/"
//   3. mkdir "$SCRATCH/ka" && tar -xzf "$SCRATCH/ka.tar.gz" -C "$SCRATCH/ka"
//      find "$SCRATCH/ka" ! -type d | wc -l    # must equal the count in step 1
//      cp "$SCRATCH/kv_capture_test.go.txt" \
//         "$SCRATCH/ka/internal/investigate/kv_capture_test.go"
//   4. cd "$SCRATCH/ka" && export PATH=$PATH:/usr/local/go/bin && \
//      KV_FIXTURE="$SCRATCH/gather_fixture.yaml" \
//      KV_FIXTURE_LOGS="$SCRATCH/gather_fixture_logs.yaml" \
//      KV_FIXTURE_CLUSTER="$SCRATCH/gather_fixture_cluster.yaml" \
//      KV_FIXTURE_REGISTRY="$SCRATCH/gather_fixture_registry.yaml" KV_GOLDEN_DIR="$SCRATCH/out" \
//      go test -count=1 -tags goldencapture -run TestCaptureVerdictGolden ./internal/investigate -v
//   5. copy back into KV: out/input.json and out/user_message.txt to
//      contract/golden/, out/system_prompt.txt to contract/,
//      out/rules_golden.json to tests/fixtures/, out/gather_go/* to
//      tests/fixtures/gather_go/, out/gather_go_logs/* to
//      tests/fixtures/gather_go_logs/, and the whole folders
//      out/gather_go_cluster and out/gather_go_registry to tests/fixtures/;
//      record the tag and commit in contract/PIN.md
//   6. check that the fixture still exercises every rule path the old
//      rules_golden.json did (each attribution rule and each Decide
//      outcome, at least once), and that the refused reads of node
//      worker-7, PVC aux-1 and the events of img/seven are still there
//   7. rm -rf "$SCRATCH" && git status --short    # must print what step 1 printed
//
// The run needs no API server. Three stand-ins replace one:
//   - no read that needs a real RESTClient runs: the harness never calls
//     scan.Evaluate (whose kubelet, CoreDNS, readyz and stats reads are
//     proxied through one; in mode cluster it calls the pure svchealth and
//     netpolicy steps Evaluate calls itself), and the wrapped CoreV1 panics
//     if anything asks for its RESTClient;
//   - a previous-log read goes to an httptest server that serves the
//     fixture's log text (the fake clientset's GetLogs always answers
//     "fake logs"), or answers 403 where the fixture sets log_refused;
//   - a PrependReactor lists events by involvedObject.name (the fake
//     ignores field selectors), and a self-check proves two workloads'
//     events reads differ.
//
// Re-run these seven steps only by the re-pin procedure in contract/PIN.md.

// kvEvent is one event, written in the fixture as [reason, message, count].
type kvEvent struct {
	Reason  string
	Message string
	Count   int32
}

func (e *kvEvent) UnmarshalJSON(b []byte) error {
	var raw []json.RawMessage
	if err := json.Unmarshal(b, &raw); err != nil {
		return err
	}
	if len(raw) != 3 {
		return fmt.Errorf("an event is [reason, message, count], got %d items", len(raw))
	}
	if err := json.Unmarshal(raw[0], &e.Reason); err != nil {
		return err
	}
	if err := json.Unmarshal(raw[1], &e.Message); err != nil {
		return err
	}
	return json.Unmarshal(raw[2], &e.Count)
}

// The kubelet text the dataset gives a node's conditions
// (kubeagent_verdict/dataset/rules.py, _MEMORY_OK and its neighbours), so
// a node describe here prints what the Python read_text prints.
var (
	kvMemoryOK     = kvCond(corev1.NodeMemoryPressure, "False", "KubeletHasSufficientMemory", "kubelet has sufficient memory available")
	kvDiskOK       = kvCond(corev1.NodeDiskPressure, "False", "KubeletHasNoDiskPressure", "kubelet has no disk pressure")
	kvDiskPressure = kvCond(corev1.NodeDiskPressure, "True", "KubeletHasDiskPressure", "kubelet has disk pressure")
	kvPIDOK        = kvCond(corev1.NodePIDPressure, "False", "KubeletHasSufficientPID", "kubelet has sufficient PID available")
	kvReadyTrue    = kvCond(corev1.NodeReady, "True", "KubeletReady", "kubelet is posting ready status")
	kvReadyFalse   = kvCond(corev1.NodeReady, "False", "KubeletNotReady", "container runtime is down")
)

// The heartbeat check's inputs: a fixed clock, scan's 40s default threshold,
// and each node's lease. A node with no lease key in the fixture gets a lease
// renewed 10s ago when its scan_reason is empty, and none when it is
// "no kubelet lease"; a NotReady node needs none, because Assess only checks
// the lease of a Ready node. A node with a lease key says what its lease
// looks like (see kvNode).
var kvNow = time.Date(2026, time.September, 26, 12, 0, 0, 0, time.UTC)

const kvThreshold = 40 * time.Second

func kvCond(t corev1.NodeConditionType, status, reason, msg string) corev1.NodeCondition {
	return corev1.NodeCondition{Type: t, Status: corev1.ConditionStatus(status), Reason: reason, Message: msg}
}

// kvScanNode is the node as the scan listed it. A node with conditions in the
// fixture gets exactly those, in order, and its cordon from unschedulable;
// scan_reason is then only what DownNodes must say (the self-check in
// kvCapture). Any other node has its Ready condition follow scan_reason and
// the other three healthy, and is never cordoned: Spec.Unschedulable stays
// false, even for a node whose fresh read finds a cordon (kvFreshNode). So
// Assess prints no MemoryPressure, DiskPressure, PIDPressure or
// SchedulingDisabled line for such a node, and the old fixtures' byte-equal
// check says nothing about them.
func kvScanNode(n kvNode) (corev1.Node, error) {
	if len(n.Conditions) > 0 {
		node := corev1.Node{ObjectMeta: metav1.ObjectMeta{Name: n.Name}}
		for _, c := range n.Conditions {
			node.Status.Conditions = append(node.Status.Conditions,
				kvCond(corev1.NodeConditionType(c.Type), c.Status, c.Reason, c.Message))
		}
		node.Spec.Unschedulable = n.Unschedulable
		return node, nil
	}
	ready := kvReadyTrue
	switch n.ScanReason {
	case "NotReady":
		ready = kvReadyFalse
	case "", "no kubelet lease":
	default:
		return corev1.Node{}, fmt.Errorf("node %s: scan_reason %q is not one the fixture supports", n.Name, n.ScanReason)
	}
	node := corev1.Node{ObjectMeta: metav1.ObjectMeta{Name: n.Name}}
	node.Status.Conditions = []corev1.NodeCondition{kvMemoryOK, kvDiskOK, kvPIDOK, ready}
	return node, nil
}

// kvLease is a node lease renewed at the given time.
func kvLease(name string, renewed time.Time) coordinationv1.Lease {
	renew := metav1.NewMicroTime(renewed)
	return coordinationv1.Lease{
		ObjectMeta: metav1.ObjectMeta{Namespace: "kube-node-lease", Name: name},
		Spec:       coordinationv1.LeaseSpec{RenewTime: &renew},
	}
}

// kvHasConditions reports whether a fixture spells out node conditions,
// so its node count is the cluster's own, not max(3, named+1).
func kvHasConditions(nodes []kvNode) bool {
	for _, n := range nodes {
		if len(n.Conditions) > 0 {
			return true
		}
	}
	return false
}

// kvFreshNode is the node a fresh describe finds, with the conditions in
// the order the Python _node_conditions gives them.
func kvFreshNode(name string, f kvFresh) (*corev1.Node, error) {
	node := &corev1.Node{ObjectMeta: metav1.ObjectMeta{Name: name}}
	node.Spec.Unschedulable = f.Unschedulable
	disk := kvDiskOK
	if f.DiskPressure {
		disk = kvDiskPressure
	}
	switch f.Ready {
	case "missing":
	case "True":
		node.Status.Conditions = []corev1.NodeCondition{kvMemoryOK, disk, kvPIDOK, kvReadyTrue}
	case "False":
		if f.ReadyReason != kvReadyFalse.Reason || f.ReadyMessage != kvReadyFalse.Message {
			return nil, fmt.Errorf("node %s: Ready=False must carry %s / %q", name, kvReadyFalse.Reason, kvReadyFalse.Message)
		}
		node.Status.Conditions = []corev1.NodeCondition{kvMemoryOK, disk, kvPIDOK, kvReadyFalse}
	default:
		return nil, fmt.Errorf("node %s: fresh ready %q is not True, False or missing", name, f.Ready)
	}
	for _, tn := range f.Taints {
		if len(tn) != 3 {
			return nil, fmt.Errorf("node %s: a taint is [key, value, effect]", name)
		}
		node.Spec.Taints = append(node.Spec.Taints, corev1.Taint{Key: tn[0], Value: tn[1], Effect: corev1.TaintEffect(tn[2])})
	}
	return node, nil
}

func kvFreshPVC(p kvPVC) *corev1.PersistentVolumeClaim {
	pvc := &corev1.PersistentVolumeClaim{ObjectMeta: metav1.ObjectMeta{Namespace: p.Namespace, Name: p.Name}}
	if p.Fresh.StorageClass != "" {
		sc := p.Fresh.StorageClass
		pvc.Spec.StorageClassName = &sc
	}
	pvc.Spec.VolumeName = p.Fresh.Volume
	pvc.Status.Phase = corev1.PersistentVolumeClaimPhase(p.Fresh.Phase)
	return pvc
}

// kvEvents turns one object's fixture events into Event objects. The names
// are zero-padded because the fake tracker lists by name: e01 before e02
// keeps the fixture's order.
func kvEvents(ns, object string, evs []kvEvent) []runtime.Object {
	var out []runtime.Object
	for i, e := range evs {
		out = append(out, &corev1.Event{
			ObjectMeta:     metav1.ObjectMeta{Namespace: ns, Name: fmt.Sprintf("%s.e%02d", object, i+1)},
			InvolvedObject: corev1.ObjectReference{Kind: "Pod", Namespace: ns, Name: object},
			Reason:         e.Reason,
			Message:        e.Message,
			Count:          e.Count,
		})
	}
	return out
}

func kvScanWorkload(fw kvWorkload) inventory.Workload {
	w := inventory.Workload{
		Namespace: fw.Namespace, Name: fw.Name, Kind: fw.Kind,
		Desired: fw.Desired, Ready: fw.Ready, Status: fw.Status, Restarts: fw.Restarts,
	}
	for _, p := range fw.Pods {
		w.Pods = append(w.Pods, inventory.PodRow{Name: p.Name, Node: p.Node})
	}
	for _, f := range fw.Findings {
		w.Findings = append(w.Findings, diagnose.Finding{
			Pod: f.Pod, Issue: f.Issue, Reason: f.Reason, Evidence: f.Evidence,
			Container: f.Container, Image: f.Image,
		})
	}
	return w
}

// kvForbidden is a refused read whose Error() is exactly msg, which is what
// redact.Error then passes through.
func kvForbidden(msg string) error {
	return &apierrors.StatusError{ErrStatus: metav1.Status{
		Status: metav1.StatusFailure, Code: http.StatusForbidden,
		Reason: metav1.StatusReasonForbidden, Message: msg,
	}}
}

// ---------------------------------------------------------------------------
// The dumps. Records are fields joined by one tab, ending in a newline. A
// value that can hold a newline goes in a block: "block\t<name>\t<N>\n",
// the N raw bytes, then "\n". tests/fixtures/gather_go/README.md is the
// contract.

type kvDump struct {
	t *testing.T
	b strings.Builder
}

func (d *kvDump) rec(fields ...string) {
	d.t.Helper()
	for _, f := range fields {
		if strings.ContainsAny(f, "\t\n\r") {
			d.t.Fatalf("dump field %q holds a tab or a line break", f)
		}
	}
	d.b.WriteString(strings.Join(fields, "\t"))
	d.b.WriteString("\n")
}

func (d *kvDump) block(name, body string) {
	d.t.Helper()
	d.rec("block", name, strconv.Itoa(len(body)))
	d.b.WriteString(body)
	d.b.WriteString("\n")
}

// kvRead is one read the gather makes, as the harness's mirror of its walk
// predicts it: the label, what kind of read it is, and the object.
type kvRead struct {
	label     string
	kind      string // events | node | pvc | log
	ns, name  string
	container string
}

// kvAction is what the gather's walk does with one candidate or one
// finding: read, deduped, ruled_out, no_object, registry, skip, or budget.
// ref is the 1-based trail index of the read it made or reused, 0 for none.
type kvAction struct {
	action string
	ref    int
}

// kvWalk mirrors gatherEvidence's loop (gather.go:57-156 at v1.24.0) check
// for check, so the dumps can say what happened to each candidate and each
// finding. The test fails unless the mirror's reads equal the trail the
// real gather returns.
func kvWalk(scoped []inventory.Workload) (reads []kvRead, cands [][]kvAction, logs [][]kvAction) {
	seenDescribe := map[string]int{}
	seenLog := map[string]int{}
	spent := 0
	add := func(r kvRead) int {
		reads = append(reads, r)
		spent++
		return len(reads)
	}
	for _, w := range scoped {
		ca := make([]kvAction, len(w.RootCauseTrace))
		la := make([]kvAction, len(w.Findings))
		for i := range ca {
			ca[i] = kvAction{action: "budget"}
		}
		for i := range la {
			la[i] = kvAction{action: "budget"}
		}
		cands, logs = append(cands, ca), append(logs, la)
		if spent >= maxToolCalls {
			continue
		}
		name := w.Name
		if len(w.Findings) > 0 {
			if p := podPart(w.Findings[0].Pod); p != "" {
				name = p
			}
		}
		add(kvRead{label: fmt.Sprintf("events %s/%s", w.Namespace, name), kind: "events", ns: w.Namespace, name: name})
		for i, h := range w.RootCauseTrace {
			if spent >= maxToolCalls {
				break
			}
			if h.Verdict == inventory.VerdictRuledOut {
				ca[i] = kvAction{action: "ruled_out"}
				continue
			}
			if h.Object == "" {
				ca[i] = kvAction{action: "no_object"}
				continue
			}
			if h.Kind != "node" && h.Kind != "pvc" {
				ca[i] = kvAction{action: "registry"}
				continue
			}
			ns := ""
			if h.Kind == "pvc" {
				ns = w.Namespace
			}
			key := h.Kind + "/" + ns + "/" + h.Object
			if ref, ok := seenDescribe[key]; ok {
				ca[i] = kvAction{action: "deduped", ref: ref}
				continue
			}
			ref := add(kvRead{label: fmt.Sprintf("describe %s %s/%s", h.Kind, ns, h.Object), kind: h.Kind, ns: ns, name: h.Object})
			seenDescribe[key] = ref
			ca[i] = kvAction{action: "read", ref: ref}
		}
		for i, f := range w.Findings {
			if spent >= maxToolCalls {
				break
			}
			pod := podPart(f.Pod)
			if !crashFamily(f.Issue) || f.Container == "" || pod == "" {
				la[i] = kvAction{action: "skip"}
				continue
			}
			key := w.Namespace + "/" + pod + "/" + f.Container
			if ref, ok := seenLog[key]; ok {
				la[i] = kvAction{action: "deduped", ref: ref}
				continue
			}
			ref := add(kvRead{label: fmt.Sprintf("log causes %s/%s container %s", w.Namespace, pod, f.Container), kind: "log", ns: w.Namespace, name: pod, container: f.Container})
			seenLog[key] = ref
			la[i] = kvAction{action: "read", ref: ref}
		}
	}
	return reads, cands, logs
}

// kvSections splits the bundle into one body per trail label. Each section
// is "== <label> ==\n<body>\n\n" (appendRead, gather.go:163-169).
func kvSections(bundle string, trail []string) ([]string, error) {
	rest := bundle
	bodies := make([]string, 0, len(trail))
	for i, label := range trail {
		head := "== " + label + " ==\n"
		if !strings.HasPrefix(rest, head) {
			return nil, fmt.Errorf("section %d does not open with %q", i+1, head)
		}
		rest = rest[len(head):]
		var end int
		if i+1 < len(trail) {
			end = strings.Index(rest, "\n\n== "+trail[i+1]+" ==\n")
		} else if strings.HasSuffix(rest, "\n\n") {
			end = len(rest) - 2
		} else {
			end = -1
		}
		if end < 0 {
			return nil, fmt.Errorf("section %d (%s) has no end", i+1, label)
		}
		bodies = append(bodies, rest[:end])
		rest = rest[end+2:]
	}
	if rest != "" {
		return nil, fmt.Errorf("%d bytes after the last section", len(rest))
	}
	return bodies, nil
}

}

func candidatesJSON(w inventory.Workload, r hypothesis.Result) []candidateJSON {
	out := make([]candidateJSON, 0, len(w.RootCauseTrace))
	for _, h := range w.RootCauseTrace {
		c := candidateJSON{Cause: h.Cause, Verdict: string(h.Verdict), Reason: h.Reason}
		for _, d := range r.Decisions {
			if d.Candidate.Cause == h.Cause {
				c.FreshReadOutcome, c.FreshReadEvidence = string(d.Outcome), d.Evidence
			}
		}
		out = append(out, c)
	}
	return out
}

func resultToJSON(r hypothesis.Result) resultJSON {
	out := resultJSON{Decided: r.Decided, Cause: r.Cause, Outcome: string(r.Outcome),
		Evidence: r.Evidence, GroupKey: r.GroupKey, GroupText: r.GroupText,
		Decisions: []decisionJSON{}}
	for _, d := range r.Decisions {
		out.Decisions = append(out.Decisions, decisionJSON{d.Candidate.Cause, string(d.Outcome), d.Evidence})
	}
	return out
}

func firstPod(w inventory.Workload) (pod, issue, image string) {
	f := w.Findings[0]
	if i := strings.IndexByte(f.Pod, '/'); i >= 0 {
		pod = f.Pod[i+1:]
	}
	return pod, f.Issue, f.Image
}

// clusterJSON, lineJSON, summaryJSON, serviceIssuesJSON and findingsJSON
// mirror kubeagent's own field shapes into the same JSON key names
// contract/golden/input.json already uses (snake_case, "type" for a
// service issue's kind). Go's json encoder sorts a map's keys, so the
// insertion order here does not matter.
func clusterJSON(c clusterhealth.ClusterHealth) map[string]any {
	nodeIssues := c.NodeIssues
	if nodeIssues == nil {
		nodeIssues = []string{}
	}
	systemIssues := c.SystemIssues
	if systemIssues == nil {
		systemIssues = []string{}
	}
	return map[string]any{
		"degraded":      c.Verdict == "Degraded",
		"nodes_ready":   c.NodesReady,
		"nodes_total":   c.NodesTotal,
		"node_issues":   nodeIssues,
		"system_issues": systemIssues,
	}
}

func lineJSON(l resources.Line) map[string]any {
	return map[string]any{
		"allocatable":  l.Allocatable,
		"requests":     l.Requests,
		"requests_pct": l.RequestsPct,
		"limits":       l.Limits,
		"limits_pct":   l.LimitsPct,
	}
}

func summaryJSON(s *resources.Summary) map[string]any {
	if s == nil {
		return map[string]any{}
	}
	return map[string]any{
		"cpu":               lineJSON(s.CPU),
		"memory":            lineJSON(s.Memory),
		"metrics_available": s.MetricsAvailable,
	}
}

func serviceIssuesJSON(issues []svchealth.Issue) []map[string]any {
	out := make([]map[string]any, 0, len(issues))
	for _, iss := range issues {
		out = append(out, map[string]any{
			"namespace": iss.Namespace,
			"name":      iss.Name,
			"type":      iss.Type,
			"detail":    iss.Detail,
		})
	}
	return out
}

// findingsJSON converts a workload's findings into the JSON shape today's
// golden uses. next_step/command are remediation's own deterministic
// suggestion for the finding's Issue, exactly as the prompt's "suggested
// fix" line reports it: explain.go's suggestionFor masks f.Pod to <pod>
// before calling remediation.For whenever the finding's pod differs from
// the owning workload's own namespace/name, and this mirrors that rule so
// input.json's command values match the prompt's masked run: lines byte
// for byte.
func findingsJSON(w inventory.Workload, findings []diagnose.Finding) []map[string]any {
	out := make([]map[string]any, 0, len(findings))
	for _, f := range findings {
		if f.Pod != "" && f.Pod != w.Namespace+"/"+w.Name {
			if i := strings.IndexByte(f.Pod, '/'); i >= 0 {
				f.Pod = f.Pod[:i+1] + "<pod>"
			} else {
				f.Pod = "<pod>"
			}
		}
		s := remediation.For(f)
		out = append(out, map[string]any{
			"issue":     f.Issue,
			"reason":    f.Reason,
			"evidence":  f.Evidence,
			"log_cause": f.LogCause,
			"next_step": s.NextStep,
			"command":   s.Command,
		})
	}
	return out
}

		t.Fatal(err)
	}
}

// kvCapture runs the pipeline over one fixture and writes its eleven dump
// files under dir/dumpName. Mode main also writes the four golden files to
// dir; mode logs checks that the log loop reached every action; modes cluster
// and registry check the lines their fixture exists to capture.
func kvCapture(t *testing.T, fixturePath, dir, dumpName string, mode kvMode) {
	dumpDir := filepath.Join(dir, dumpName)
	if err := os.MkdirAll(dumpDir, 0o755); err != nil {
		t.Fatal(err)
	}
	ctx := context.Background()

	raw, err := os.ReadFile(fixturePath)
	if err != nil {
		t.Fatal(err)
	}
	var fx kvFixture
	if err := yaml.UnmarshalStrict(raw, &fx); err != nil {
		t.Fatalf("loading %s: %v", fixturePath, err)
	}
	if fx.PlatformLine != "" {
		t.Fatal("platform_line must be empty: the capture has no platform facts to print it from")
	}
	asJSON, err := yaml.YAMLToJSON(raw)
	if err != nil {
		t.Fatal(err)
	}
	var dump0 bytes.Buffer
	if err := json.Indent(&dump0, asJSON, "", "  "); err != nil {
		t.Fatal(err)
	}
	dump0.WriteString("\n")
	writeFile(t, dumpDir, "00-fixture.json", dump0.Bytes())

	summary := &resources.Summary{
		CPU: resources.Line{Allocatable: fx.Summary.CPU.Allocatable, Requests: fx.Summary.CPU.Requests,
			RequestsPct: fx.Summary.CPU.RequestsPct, Limits: fx.Summary.CPU.Limits, LimitsPct: fx.Summary.CPU.LimitsPct},
		Memory: resources.Line{Allocatable: fx.Summary.Memory.Allocatable, Requests: fx.Summary.Memory.Requests,
			RequestsPct: fx.Summary.Memory.RequestsPct, Limits: fx.Summary.Memory.Limits, LimitsPct: fx.Summary.Memory.LimitsPct},
		MetricsAvailable: fx.Summary.MetricsAvailable,
	}
	var serviceIssues []svchealth.Issue
	for _, s := range fx.ServiceIssues {
		serviceIssues = append(serviceIssues, svchealth.Issue{Namespace: s.Namespace, Name: s.Name, Type: s.Type, Detail: s.Detail})
	}
	// The service lists belong to mode cluster, where the harness computes
	// the issues from them; the hand-written service_issues belong to the
	// others.
	if mode == kvModeCluster && len(fx.ServiceIssues) > 0 {
		t.Fatal("service_issues must be empty in mode cluster: the harness computes them from services")
	}
	if mode != kvModeCluster && (len(fx.Services) > 0 || len(fx.EndpointSlices) > 0 || len(fx.Backends) > 0 || len(fx.NetworkPolicies) > 0) {
		t.Fatal("services, endpoint_slices, backends and network_policies belong to mode cluster only")
	}

	// The cluster: scan-time nodes and leases for Assess, fresh objects for
	// the fake, and the refused reads for the reactors.
	var (
		objects       []runtime.Object
		scanNodes     []corev1.Node
		leases        []coordinationv1.Lease
		refusedNodes  = map[string]string{}
		refusedPVCs   = map[string]string{}
		refusedEvents = map[string]string{}
		fxNodes       = map[string]kvNode{}
		fxPVCs        = map[string]kvPVC{}
	)
	nodes := append([]kvNode(nil), fx.Nodes...)
	sort.Slice(nodes, func(i, j int) bool { return nodes[i].Name < nodes[j].Name })
	for _, n := range nodes {
		fxNodes[n.Name] = n
		sn, err := kvScanNode(n)
		if err != nil {
			t.Fatal(err)
		}
		scanNodes = append(scanNodes, sn)
		switch n.Lease {
		case "":
			if n.ScanReason == "" { // today's rule, unchanged for the old fixtures
				leases = append(leases, kvLease(n.Name, kvNow.Add(-10*time.Second)))
			}
		case "renewed":
			leases = append(leases, kvLease(n.Name, kvNow.Add(-time.Duration(n.LeaseAgeMs)*time.Millisecond)))
		case "no_renew":
			leases = append(leases, coordinationv1.Lease{ObjectMeta: metav1.ObjectMeta{Namespace: "kube-node-lease", Name: n.Name}})
		case "missing":
		default:
			t.Fatalf("node %s: lease %q is not renewed, no_renew or missing", n.Name, n.Lease)
		}
		switch n.Fresh.How {
		case "read":
			fn, err := kvFreshNode(n.Name, n.Fresh)
			if err != nil {
				t.Fatal(err)
			}
			objects = append(objects, fn)
		case "read_failed":
			refusedNodes[n.Name] = n.Fresh.Message
		default:
			t.Fatalf("node %s: fresh how %q is not read or read_failed", n.Name, n.Fresh.How)
		}
	}
	var pvcIssues []pvchealth.Issue
	for _, p := range fx.PVCs {
		fxPVCs[p.Namespace+"/"+p.Name] = p
		pvcIssues = append(pvcIssues, pvchealth.Issue{Namespace: p.Namespace, Name: p.Name, Phase: "Pending",
			Reason: p.ScanReason, Detail: "waiting for a volume", StorageClass: p.Fresh.StorageClass})
		switch p.Fresh.How {
		case "read":
			objects = append(objects, kvFreshPVC(p))
		case "read_failed":
			refusedPVCs[p.Namespace+"/"+p.Name] = p.Fresh.Message
		default:
			t.Fatalf("pvc %s/%s: fresh how %q is not read or read_failed", p.Namespace, p.Name, p.Fresh.How)
		}
	}

	// The workloads, their pods' claims, events and log text.
	var workloads []inventory.Workload
	fxByKey := map[string]kvWorkload{}
	podPVCs := map[string][]string{}
	logText := map[string]string{}       // "ns/pod/container"
	refusedLogs := map[string]bool{}     // "ns/pod/container"
	logFinding := map[string]kvFinding{} // "ns/pod/container" -> its first finding
	var allPods []corev1.Pod             // every fixture pod, for the service and policy steps
	podLabels := map[string]map[string]string{}
	for _, fw := range fx.Workloads {
		if _, dup := fxByKey[fw.key()]; dup {
			t.Fatalf("workload %s appears twice", fw.key())
		}
		fxByKey[fw.key()] = fw
		workloads = append(workloads, kvScanWorkload(fw))
		eventsPod := fw.Name
		if len(fw.Findings) > 0 {
			eventsPod = podPart(fw.Findings[0].Pod)
		}
		if fw.Pod != eventsPod {
			t.Fatalf("%s: pod is %q, but its events pod is %q", fw.key(), fw.Pod, eventsPod)
		}
		for _, p := range fw.Pods {
			if len(p.Claims) > 0 {
				podPVCs[fw.Namespace+"/"+p.Name] = append([]string(nil), p.Claims...)
			}
			pod := corev1.Pod{ObjectMeta: metav1.ObjectMeta{Namespace: fw.Namespace, Name: p.Name, Labels: p.Labels}}
			pod.Spec.NodeName = p.Node
			status := corev1.ConditionFalse
			if p.Ready {
				status = corev1.ConditionTrue
			}
			pod.Status.Conditions = []corev1.PodCondition{{Type: corev1.PodReady, Status: status}}
			allPods = append(allPods, pod)
			podLabels[fw.Namespace+"/"+p.Name] = p.Labels
		}
		for _, f := range fw.Findings {
			if !strings.HasPrefix(f.Pod, fw.Namespace+"/") {
				t.Fatalf("%s: finding pod %q is outside the namespace", fw.key(), f.Pod)
			}
			reads := crashFamily(f.Issue) && f.Container != ""
			if reads != (f.LogRead != "") || (!reads && (f.Log != "" || f.LogRefused)) {
				t.Fatalf("%s: %s on %s: log, log_refused and log_read belong on a crash-family finding with a container, and only there", fw.key(), f.Issue, f.Pod)
			}
			if f.LogRefused && f.Log != "" {
				t.Fatalf("%s: %s on %s: a refused log read serves no log", fw.key(), f.Issue, f.Pod)
			}
			if reads {
				key := fw.Namespace + "/" + podPart(f.Pod) + "/" + f.Container
				if prev, seen := logFinding[key]; seen && (prev.Log != f.Log || prev.LogRefused != f.LogRefused || prev.LogRead != f.LogRead) {
					t.Fatalf("%s: two findings on %s disagree on its log", fw.key(), key)
				}
				logFinding[key] = f
				logText[key] = f.Log
				if f.LogRefused {
					refusedLogs[key] = true
				}
			}
		}
		if fw.EventsFailed != "" {
			if len(fw.Events) > 0 {
				t.Fatalf("%s: events and events_failed are both set", fw.key())
			}
			refusedEvents[fw.Namespace+"/"+fw.Pod] = fw.EventsFailed
		}
		objects = append(objects, kvEvents(fw.Namespace, fw.Pod, fw.Events)...)
		for _, d := range fw.DecoyEvents {
			if d.Object == fw.Pod {
				t.Fatalf("%s: a decoy names the events pod itself", fw.key())
			}
			objects = append(objects, kvEvents(fw.Namespace, d.Object, d.Events)...)
		}
	}

	// The fake clientset and its reactors.
	cs := fake.NewSimpleClientset(objects...)
	served := map[string][]corev1.Event{} // "ns/name" -> what the events reactor returned
	cs.PrependReactor("list", "events", func(a k8stesting.Action) (bool, runtime.Object, error) {
		name, ok := a.(k8stesting.ListAction).GetListRestrictions().Fields.RequiresExactMatch("involvedObject.name")
		if !ok {
			t.Errorf("an events list without an involvedObject.name selector")
			return true, nil, fmt.Errorf("no involvedObject.name selector")
		}
		key := a.GetNamespace() + "/" + name
		if msg, refused := refusedEvents[key]; refused {
			return true, nil, kvForbidden(msg)
		}
		obj, err := cs.Tracker().List(corev1.SchemeGroupVersion.WithResource("events"),
			corev1.SchemeGroupVersion.WithKind("Event"), a.GetNamespace())
		if err != nil {
			return true, nil, err
		}
		out := &corev1.EventList{}
		for _, e := range obj.(*corev1.EventList).Items {
			if e.InvolvedObject.Name == name {
				out.Items = append(out.Items, e)
			}
		}
		served[key] = out.Items
		return true, out, nil
	})
	cs.PrependReactor("get", "nodes", func(a k8stesting.Action) (bool, runtime.Object, error) {
		if msg, refused := refusedNodes[a.(k8stesting.GetAction).GetName()]; refused {
			return true, nil, kvForbidden(msg)
		}
		return false, nil, nil
	})
	cs.PrependReactor("get", "persistentvolumeclaims", func(a k8stesting.Action) (bool, runtime.Object, error) {
		if msg, refused := refusedPVCs[a.GetNamespace()+"/"+a.(k8stesting.GetAction).GetName()]; refused {
			return true, nil, kvForbidden(msg)
		}
		return false, nil, nil
	})

	logsServed := 0
	srv := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		parts := strings.Split(strings.TrimPrefix(r.URL.Path, "/api/v1/namespaces/"), "/")
		if len(parts) != 4 || parts[1] != "pods" || parts[3] != "log" {
			t.Errorf("unexpected request to the log server: %s", r.URL.Path)
			http.NotFound(w, r)
			return
		}
		q := r.URL.Query()
		if q.Get("previous") != "true" || q.Get("tailLines") != "25" {
			t.Errorf("log read %s asks for previous=%q tailLines=%q", r.URL.Path, q.Get("previous"), q.Get("tailLines"))
		}
		key := parts[0] + "/" + parts[2] + "/" + q.Get("container")
		text, ok := logText[key]
		if !ok {
			t.Errorf("no fixture log for %s/%s container %s", parts[0], parts[2], q.Get("container"))
			http.NotFound(w, r)
			return
		}
		logsServed++
		if refusedLogs[key] {
			// The API server's own answer to a refused read: a 403 with a
			// Status body, which client-go turns into a Forbidden error.
			status := metav1.Status{
				TypeMeta: metav1.TypeMeta{Kind: "Status", APIVersion: "v1"},
				Status:   metav1.StatusFailure, Code: http.StatusForbidden, Reason: metav1.StatusReasonForbidden,
				Message: fmt.Sprintf("pods %q is forbidden: cannot get resource \"pods/log\"", parts[2]),
			}
			body, err := json.Marshal(status)
			if err != nil {
				t.Errorf("encoding the refusal of %s: %v", key, err)
			}
			w.Header().Set("Content-Type", "application/json")
			w.WriteHeader(http.StatusForbidden)
			_, _ = w.Write(body)
			return
		}
		w.Header().Set("Content-Type", "text/plain")
		_, _ = w.Write([]byte(text))
	}))
	defer srv.Close()
	real, err := kubernetes.NewForConfig(&rest.Config{Host: srv.URL})
	if err != nil {
		t.Fatal(err)
	}
	client := kvLogClient{Clientset: cs, core: kvLogCore{CoreV1Interface: cs.CoreV1(), real: real.CoreV1()}}

	// The scan, in scan.go's order (internal/scan/scan.go:749, :793, :815-818).
	assessed := append([]inventory.Workload(nil), workloads...)
	sort.Slice(assessed, func(i, j int) bool {
		a, b := assessed[i], assessed[j]
		if a.Namespace != b.Namespace {
			return a.Namespace < b.Namespace
		}
		if a.Name != b.Name {
			return a.Name < b.Name
		}
		return a.Kind < b.Kind
	})
	health := clusterhealth.Assess(scanNodes, clusterhealth.Heartbeat{Leases: leases, Now: kvNow, Threshold: kvThreshold}, nil, assessed)
	prioritized := inventory.Prioritize(workloads, inventory.Opts{})
	wl := prioritized.Workloads
	if mode == kvModeCluster {
		// scan.go:752-754 and :813: the Service issues, the cause each broken
		// one gets from the down nodes, and the NetworkPolicy hint, which runs
		// on the prioritized workloads before rootcause.
		var svcs []corev1.Service
		for _, s := range fx.Services {
			svc := corev1.Service{ObjectMeta: metav1.ObjectMeta{Namespace: s.Namespace, Name: s.Name, Annotations: s.Annotations}}
			svc.Spec.Type = corev1.ServiceType(s.Type)

			// differ by design.

		}
	}
	if cuts == 1 && strings.Contains(bundle, "\n"+truncationMarker+"\n") {
		has["read cut on a line"] = true
	}
	for _, need := range []string{
		"read of two crash-family issues", "deduped", "skip: not crash-family", "skip: no container",
		"budget on a crash-family finding", "log cause", "log with no classifiable output",
		"read of a log holding an address", "refused log read", "no previous-instance log",
		"read cut on a line",
	} {
		if !has[need] {
			t.Fatalf("the log fixture never reaches %q", need)
		}
	}
	logReads := 0
	for _, r := range mirror {
		if r.kind == "log" {
			logReads++
		}
	}
	t.Logf("wrote 11 dumps under %s; %d workloads in the fixture, %d shown, %d in scope, %d reads, %d of them log reads",
		dumpDir, len(fx.Workloads), len(wl), len(scoped), len(trail), logReads)
}

// kvCheckCluster fails unless the prompt shows every line the CLUSTER
// fixture exists to capture.
func kvCheckCluster(t *testing.T, prompt string) {
	t.Helper()
	for _, s := range []string{
		"Cluster health (P1): DEGRADED — 11/16 nodes Ready.",
		"  node n01 NotReady: KubeletNotReady — ", "…\n",
		"  node n02 NotReady: NodeStatusUnknown — Kubelet stopped posting node status.",
		"  node n03 NotReady: PLEG is not healthy",
		"  node n04 NotReady\n",
		"  node n05 no kubelet lease", "  node n06 no kubelet lease",
		"(lease 40s stale)", "(lease 43s stale)", "(lease 1m35s stale)",
		"(lease 1h0m0s stale)", "(lease 1h1m40s stale)",
		"  node n13 MemoryPressure\n  node n13 PIDPressure\n",
		"  node n14 SchedulingDisabled",
		"  node n15 DiskPressure\n  node n15 NotReady: KubeletNotReady — container runtime is down\n  node n15 SchedulingDisabled\n",
		"  system kube-system/ks-cron Last run failed",
		"network policy: pods selected by allow-web, deny-web",
	} {
		if !strings.Contains(prompt, s) {
			t.Errorf("cluster prompt is missing %q", s)
		}
	}
	for _, s := range []string{"  node n12 ", "  node n16 ", "other-ns-policy"} {
		if strings.Contains(prompt, s) {
			t.Errorf("cluster prompt shows %q, which must not print", s)
		}
	}
	if got := strings.Count(prompt, "\n  system kube-system/"); got < 11 {
		t.Errorf("%d system lines, want every flagged kube-system workload (at least 11)", got)
	}
}

// kvCheckRegistry fails unless the candidate dump holds both registry
// verdicts of the attribution pass (the pull threshold met, and not met) and
// the decide dump holds every outcome the rules give a registry candidate.
func kvCheckRegistry(t *testing.T, candidates, decide string) {
	t.Helper()
	for _, s := range []string{
		"only workload failing to pull from this host; threshold is 2",
		"workloads failing to pull from this host clear the threshold of 2",
	} {
		if !strings.Contains(candidates, s) {
			t.Errorf("registry candidate dump is missing %q", s)
		}
	}
	for _, s := range []string{
		"\tconfirmed\t", "\tunverified\t", "\trefuted\t",
		"a pull event shows a connection error: dial tcp",
		"a pull event shows an auth error: unauthorized; that can be one image or the whole host",
		"a pull event shows an image error: manifest unknown; this pull fails for this image, not the host",
		"no pull event names the failure; events may have aged out",
		"events of the pulling pod were not read",
	} {
		if !strings.Contains(decide, s) {
			t.Errorf("registry decide dump is missing %q", s)
		}
	}
}
